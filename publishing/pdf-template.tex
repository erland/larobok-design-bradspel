% Kapitelstart som i EPUB: nummer och rubrik på två centrerade, kompakta rader.
\newcommand{\bookchapter}[2]{%
  \clearpage
  \phantomsection
  \addcontentsline{toc}{chapter}{Kapitel #1: #2}%
  \markboth{Kapitel #1: #2}{Kapitel #1: #2}%
  \thispagestyle{plain}%
  \vspace*{10mm}%
  \begin{center}
    {\fontsize{16}{19}\selectfont\bfseries Kapitel #1\par}
    \vspace{1.2mm}
    {\fontsize{15}{18}\selectfont\bfseries #2\par}
  \end{center}
  \vspace{7mm}%
}


\newcommand{\bookintro}[1]{%
  \clearpage
  \phantomsection
  \addcontentsline{toc}{chapter}{#1}%
  \markboth{#1}{#1}%
  \thispagestyle{plain}%
  \vspace*{10mm}%
  \begin{center}
    {\fontsize{16}{19}\selectfont\bfseries #1\par}
  \end{center}
  \vspace{7mm}%
}
